import sys
 
input = sys.stdin.readline
def solve():
    s = input().strip()
 
    c = 0
    for i in range(len(s)):
        for j in range(i, len(s)):
            for k in range(j, len(s)):
                if s[i] == "Q" and s[j] == "A" and s[k] == "Q":
                    c += 1
 
    return c
def main():
 
    print(solve())
 
if __name__ == "__main__":
    main()
