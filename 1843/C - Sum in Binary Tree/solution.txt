import sys
 
input = sys.stdin.readline
 
def solve():
    n = int(input())
    ans = n
    if n&1 == 0:
        while n>1:
            n = n//2
            ans += n
    elif n&1 != 0:
        n = n-1 # become even number
        while n>1:
            n = n//2
            ans += n
    return ans
 
def main():
    t = int(input())
    for _ in range(t):
        print(solve())
 
if __name__ == "__main__":
    main()
