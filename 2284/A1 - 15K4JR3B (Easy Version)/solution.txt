import sys
 
input = sys.stdin.readline
 
def solve():
    n = int(input())
    x,y = 0,0
    if n %2 ==0:
        x,y = n//2,n//2
    else:
        x,y = n//2,n//2+1
    
    return x*y
    
        
        
 
def main():
   
    t = int(input())
    for _ in range(t):
        print(solve())
if __name__ == "__main__":
    main()
