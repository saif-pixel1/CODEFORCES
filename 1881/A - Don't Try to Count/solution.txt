import sys
 
input = sys.stdin.readline
 
def solve(x,s,n,m):
    count = 0
    for i in range(10):
        if s in x:
            return count
        if len(x) > m+n -1:
            return -1
 
        x = x + x
        count += 1
    return -1
 
 
def main():
    t = int(input())
    for _ in range(t):
        n,m = map(int,input().split(" "))
        x = str(input().strip())
        s = str(input().strip())
        print(solve(x,s,n,m))
 
if __name__ == "__main__":
    main()
