import sys
 
input = sys.stdin.readline
 
def solve():
    n = int(input())
    s = input().strip()
 
    stack = []
    up = []
    for i in range(n):
        if s[i] == '1':
            stack.append(i + 1)
        elif s[i] == '2':
            if stack:
                stack.pop()
                up.append(i+1)
            
            
    ans = sorted(up+stack)
 
    print(len(ans))
    print(*ans)
 
def main():
    t = int(input())
 
    for _ in range(t):
        solve()
 
if __name__ == "__main__":
    main()
