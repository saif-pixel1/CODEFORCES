import sys
 
input = sys.stdin.readline
def solve():
    x,n = map(int, input().split())
    if n & 1 == 1:
        # odd 
        return x
    elif n & 1 == 0:
        # even 
        return 0
        
        
 
 
def main():
    
    t = int(input())
    for _ in range(t):
        print(solve())
 
if __name__ == "__main__":
    main()
