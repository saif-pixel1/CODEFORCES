import sys
 
input = sys.stdin.readline
def solve():
    n = int(input())
    if n == 0 or n == 1:
        return n
    if n%2 != 0:
        count = n//2
        count += 1
        return count
    else:
        count = n//2
        return count
 
def main():
 
    t = int(input())
    for _ in range(t):
        print(solve())
 
if __name__ == "__main__":
    main()
