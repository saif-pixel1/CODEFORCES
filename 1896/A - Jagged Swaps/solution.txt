import sys
 
input = sys.stdin.readline
 
def sol(a, n):
    l = 1
    r = 2
 
    while r < n:
        if a[l - 1] < a[l] and a[l] > a[r]:
            a[l], a[r] = a[r], a[l]
 
            l = max(1, l - 1)
            r = l + 1
        else:
            l += 1
            r += 1
 
    if a == sorted(a):
        print("YES")
    else:
        print("NO")
 
def solve(a, n):
    cond = True
    while cond:
        cond = False
        for i in range(1, n - 1):
            if a[i - 1] < a[i] and a[i] > a[i + 1]:
                a[i], a[i + 1] = a[i + 1], a[i]
                cond = True
 
    if a == sorted(a):
        print("YES")
    else:
        print("NO")
 
def main():
    t = int(input())
 
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        #solve(a, n)
        sol(a, n)
 
if __name__ == "__main__":
    main()
