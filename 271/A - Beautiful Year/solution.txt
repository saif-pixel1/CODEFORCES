import sys
 
input = sys.stdin.readline
def solve():
    yr = int(input())
    yr = yr+1
    while True:
        if len(set(str(yr)))==4:
            print(yr)
            break
        yr+=1
        
        
def main():
 
    t = 1
    for _ in range(t):
        solve()
if __name__ == "__main__":
    main()
