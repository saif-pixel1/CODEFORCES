import sys
 
input = sys.stdin.readline
 
def check(x, arr, d):
    n = len(arr)
    b= []
 
    i = 0
    while i < n - 1:
        if d[i] > x:
            b.append(i)
        i += 1
 
    if len(b) > 2:
        return False
 
    half = x // 2
 
    if len(b) == 0:
        return True
 
    if len(b) == 2:
        q1 = b[0]
        q2 = b[1]
 
        l = q1 + 1
        r = q2
 
        i = l
        while i < r:
            if d[i] > half:
                return False
            i += 1
 
        if abs(arr[q1] - 2 * arr[r]) > x:
            return False
 
        if abs(2 * arr[l] - arr[q2 + 1]) > x:
            return False
 
        return True
 
    q = b[0]
 
    l = q + 1
    r = l
    mx = 0
 
    while r < n:
        if r > l:
            mx = max(mx, d[r - 1])
 
        if mx > half:
            break
 
        if abs(arr[q] - 2 * arr[r]) <= x:
            if r == n - 1 or abs(2 * arr[l] - arr[r + 1]) <= x:
                return True
 
        r += 1
 
    r = q
    l = r
    mx = 0
 
    while l >= 0:
        if l < r:
            mx = max(mx, d[l])
 
        if mx > half:
            break
 
        if l == 0 or abs(arr[l - 1] - 2 * arr[r]) <= x:
            if abs(2 * arr[l] - arr[q + 1]) <= x:
                return True
 
        l -= 1
 
    return False
 
def solve():
    n = int(input())
    arr = list(map(int, input().split()))
 
    if n == 1:
        return 0
 
    d = []
 
    i = 0
    while i < n - 1:
        d.append(abs(arr[i] - arr[i + 1]))
        i += 1
 
    low = 0
    high = max(arr)
 
    while low < high:
        mid = (low + high) // 2
 
        if check(mid, arr, d):
            high = mid
        else:
            low = mid + 1
 
    return low
 
def main():
    t = int(input())
 
    for _ in range(t):
        print(solve())
 
if __name__ == "__main__":
    main()
