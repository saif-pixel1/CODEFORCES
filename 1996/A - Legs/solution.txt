import sys
input = sys.stdin.readline
 
def solve():
    n = int(input())
    p = 0
    c=0
    while p+4 <= n:
        p += 4
        c += 1
    while p+2 <= n:
        p += 2
        c += 1
 
    return c
 
def main():
 
    t = int(input())
    for _ in range(t):
        print(solve())
 
if __name__ == "__main__":
    main()
